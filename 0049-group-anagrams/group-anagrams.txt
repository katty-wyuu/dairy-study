/**
 * @param {string[]} strs
 * @return {string[][]}
 */
var groupAnagrams = function (strs) {
    const map = new Map();
    for(let i of strs){
        let item = [...i].sort().join('');
        if(!map.has(item)){
            map.set(item, [])
        }
        map.get(item).push(i)
    }
    return Array.from(map.values())
};
